# =========================
# FORMATADOR (M)
# =========================
def formatador(valor, pos):
    if valor == 0:
        return "0"
    elif valor >= 1_000_000:
        return f"{int(valor/1_000_000)}M"
    else:
        return ""

# =========================
# FIGURA
# =========================
plt.rcParams["font.family"] = vc.FONTE_FAMILIA

fig, ax = plt.subplots(figsize=vc.FIGSIZE, dpi=vc.DPI)

fig.patch.set_facecolor(vc.FUNDO)
ax.set_facecolor(vc.FUNDO)

plt.subplots_adjust(
    left=vc.AXES["left"],
    right=vc.AXES["right"],
    top=vc.AXES["top"],
    bottom=vc.AXES["bottom"]
)

# =========================
# SCATTER
# =========================
ax.scatter(
    valor_operacao,
    juros,
    color=vc.VERMELHO,

# =========================

    "Operação por juros",
    fontsize=vc.FONTE_TITULO,
    color=vc.TEXTO
)

# =========================
# EIXOS
# =========================
ax.set_xlabel("Valor da operação", fontsize=vc.FONTE_ROTULO)

# =========================
# GRID (igual referência)
# =========================
ax.grid(True, which='major', linestyle='-', linewidth=0.5, alpha=0.3)

# =========================
# EIXO X FORMATADO
# =========================
ax.set_xlim(-10_000_000, 310_000_000)

# =========================
# EXPORTAÇÃO SVG
